5/// Registers `body` to run in the registration phase.
6///
7/// Emits the `linkme` slice element directly instead of `#[storm::register]` to avoid a nested
8/// proc-macro expansion.
9pub(crate) fn registration(
10    name: &Ident,
11    body: proc_macro2::TokenStream,
12) -> proc_macro2::TokenStream {
13    quote! {
14        #[allow(non_upper_case_globals)]
15        #[storm::linkme::distributed_slice(storm::registry::__REGISTRATION)]
16        #[linkme(crate = storm::linkme)]
17        static #name: fn() = || { #body };
18    }
19}

21/// Body of a function returning the `storm::CtxVar<#ty>` singleton.
22///
23/// The var must also be touched in a `registration` so it is allocated before any `Ctx` exists.
24pub(crate) fn ctx_var(ty: proc_macro2::TokenStream) -> proc_macro2::TokenStream {
25    quote! {
26        static V: storm::LazyCtxVar<#ty> = storm::LazyCtxVar::new();
27        V.get()
28    }
29}

31pub fn register(attr: TokenStream, item: TokenStream) -> TokenStream {
32    let crate_path: Path = match syn::parse::<syn::Meta>(attr) {
33        Ok(syn::Meta::NameValue(nv)) if nv.path.is_ident("crate") => match nv.value {
34            syn::Expr::Path(p) => p.path,
35            _ => {
36                return syn::Error::new(nv.value.span(), "expected path")
37                    .to_compile_error()
38                    .into();
39            }
40        },
41        Ok(_) | Err(_) => syn::parse_quote!(storm), // default / no attr
42    };
43
44    let func = parse_macro_input!(item as ItemFn);
45    if !func.sig.inputs.is_empty() {
46        return syn::Error::new(func.sig.inputs.span(), "function must take no arguments")
47            .to_compile_error()
48            .into();
49    }
50
51    let fn_name = &func.sig.ident;
52    let reg_ident = format_ident!("__{}", fn_name.to_string().to_uppercase());
53
54    quote! {
55        #func
56        #[#crate_path::linkme::distributed_slice(#crate_path::registry::__REGISTRATION)]
57        #[linkme(crate = #crate_path::linkme)]
58        static #reg_ident: fn() = #fn_name;
59    }
60    .into()
61}
