1//! Dry-run of a cascaded removal, executable in batches.
2//!
3//! [DeletePlan] walks the foreign keys (`on_delete = "cascade" | "set_null" | "restrict"`)
4//! and the business rules ([on_remove_rule](crate::on_remove_rule)) from a root entity without modifying anything. The plan can be inspected (graph, counts,
5//! restrictions) and then executed leaves first, in multiple transactions, so the queue lock
6//! can be released between batches and the progress reported.
7//!
8//! Every step is idempotent and re-validated when executed: a step is skipped when its entity
9//! is gone or when any link of its foreign key chain up to the root no longer holds. Rows
10//! created after the plan are still cascaded when their parent is removed.
11

31type PlanFn = for<'a> fn(
32    trx: &'a mut CtxTransaction<'_>,
33    key: &'a (dyn Any + Send + Sync),
34    planner: &'a mut DeletePlanner,
35) -> BoxFuture<'a, Result<()>>;
36
37/// Planning handlers by entity type, in registration order.
38static PLANNERS: InitCell<FxHashMap<TypeId, Vec<PlanFn>>> =
39    InitCell::new(FxHashMap::with_hasher(FxBuildHasher));
40
41/// Registers a planning handler called when an entity `P` is planned for removal.
42pub fn on_planning<P: 'static>(f: PlanFn) {
43    PLANNERS
44        .get_mut()
45        .entry(TypeId::of::<P>())
46        .or_default()
47        .push(f);
48}
49
50#[inline]
51fn planners_of(ty: TypeId) -> &'static [PlanFn] {
52    PLANNERS.get().get(&ty).map_or(&[], Vec::as_slice)
53}

55#[derive(Clone, Copy, Debug, Eq, Hash, PartialEq)]
56pub enum DeleteAction {
57    Remove,
58    SetNull,
59}

61/// A restriction blocking the plan.
62#[derive(Clone, Debug)]
63pub enum DeleteRestriction {
64    /// A foreign key with `on_delete = "restrict"`.
65    ForeignKey {
66        /// Index of the blocked node in the plan.
67        node: usize,
68        entity: &'static str,
69        field: &'static str,
70        parent: &'static str,
71        key: String,
72        count: usize,
73    },
74
75    /// A business rule registered with [on_remove_rule](crate::on_remove_rule).
76    Rule {
77        /// Index of the blocked node in the plan.
78        node: usize,
79        entity: &'static str,
80        key: String,
81        reason: String,
82    },
83}
84
85impl DeleteRestriction {
86    #[inline]
87    pub fn node(&self) -> usize {
88        match self {
89            Self::ForeignKey { node, .. } | Self::Rule { node, .. } => *node,
90        }
91    }
92
93    pub fn to_error(&self) -> Error {
94        match self {
95            Self::ForeignKey {
96                entity,
97                field,
98                parent,
99                key,
100                count,
101                ..
102            } => Error::ForeignKeyRestricted {
103                entity,
104                field,
105                parent,
106                key: key.clone(),
107                count: *count,
108            },
109            Self::Rule {
110                entity,
111                key,
112                reason,
113                ..
114            } => Error::RemoveRestricted {
115                entity,
116                key: key.clone(),
117                reason: reason.clone(),
118            },
119        }
120    }
121}

123/// A node of the plan graph.
124pub struct DeletePlanNode<'a> {
125    pub index: usize,
126    pub parent: Option<usize>,
127    pub depth: usize,
128    pub entity: &'static str,
129    /// The foreign key field linking the node to its parent (`None` for the root).
130    pub field: Option<&'static str>,
131    pub action: DeleteAction,
132    step: &'a dyn PlanStep,
133}

135impl DeletePlanNode<'_> {
136    #[inline]
137    pub fn key(&self) -> &dyn Debug {
138        self.step.key_debug()

141    /// The typed key, i.e. `key_as::<<E as Entity>::Key>()`.

154pub struct DeletePlan {
155    nodes: Vec<Node>,
156    restrictions: Vec<DeleteRestriction>,
157    done: usize,
158}
159
160impl DeletePlan {
161    /// Builds the plan of the removal of `key` without modifying anything.
162    pub async fn new<E>(trx: &mut CtxTransaction<'_>, key: E::Key) -> Result<Self>
163    where
164        E: EntityRemove + CtxTypeInfo,
165        ProviderContainer: LoadAll<E, (), E::Tbl>,
166        for<'b> TransactionProvider<'b>: Delete<E>,
167    {
168        let mut plan = DeletePlan {
169            nodes: Vec::new(),
170            restrictions: Vec::new(),
171            done: 0,
172        };
173
174        if entity_of::<E>(trx, &key).await?.is_none() {
175            return Ok(plan);
176        }
177
178        let mut planner = DeletePlanner {
179            node: 0,
180            pending: Vec::new(),
181            restrictions: Vec::new(),
182            visited: FxHashMap::default(),
183        };
184
185        planner.visit::<E>(&key);
186        plan.nodes.push(Node {
187            parent: None,
188            depth: 0,
189            step: Box::new(RootStep::<E>(key)),
190        });
191
192        // parents are always before their children, so the plan executes in reverse order.
193        let mut i = 0;
194
195        while i < plan.nodes.len() {
196            let node = &plan.nodes[i];
197
198            if node.step.action() == DeleteAction::Remove {
199                let ty = node.step.entity_type();
200                planner.node = i;
201
202                for rule in remove_rule::rules_of(ty) {
203                    if let Some(reason) = rule.check(trx, node.step.key_any()).await? {
204                        planner.restrictions.push(DeleteRestriction::Rule {
205                            node: i,
206                            entity: rule.entity,
207                            key: format!("{:?}", node.step.key_debug()),
208                            reason,
209                        });
210                    }
211                }
212
213                for f in planners_of(ty) {
214                    f(trx, node.step.key_any(), &mut planner).await?;
215                }
216
217                let depth = node.depth + 1;
218
219                plan.nodes
220                    .extend(planner.pending.drain(..).map(|step| Node {
221                        parent: Some(i),
222                        depth,
223                        step,
224                    }));
225            }
226
227            i += 1;
228        }
229
230        plan.restrictions = planner.restrictions;
231        Ok(plan)
232    }
233
234    #[inline]
235    pub fn done(&self) -> usize {
236        self.done
237    }
238
239    #[inline]
240    pub fn is_done(&self) -> bool {
241        self.done == self.nodes.len()
242    }
243
244    #[inline]
245    pub fn is_empty(&self) -> bool {
246        self.nodes.is_empty()
247    }
248
249    #[inline]
250    pub fn is_restricted(&self) -> bool {
251        !self.restrictions.is_empty()
252    }
253
254    /// Total number of steps.
255    #[inline]
256    pub fn len(&self) -> usize {
257        self.nodes.len()
258    }
259
260    /// The graph of the plan. Parents are always listed before their children.
261    pub fn nodes(&self) -> impl ExactSizeIterator<Item = DeletePlanNode<'_>> {
262        self.nodes
263            .iter()
264            .enumerate()
265            .map(|(index, n)| DeletePlanNode {
266                index,
267                parent: n.parent,
268                depth: n.depth,
269                entity: n.step.entity(),
270                field: n.step.field(),
271                action: n.step.action(),
272                step: &*n.step,
273            })
274    }
275
276    #[inline]
277    pub fn remaining(&self) -> usize {
278        self.nodes.len() - self.done
279    }
280
281    #[inline]
282    pub fn restrictions(&self) -> &[DeleteRestriction] {
283        &self.restrictions
284    }
285
286    /// Number of steps by entity and action, in order of first appearance.
287    pub fn summary(&self) -> Vec<(&'static str, DeleteAction, usize)> {
288        let mut out: Vec<(&'static str, DeleteAction, usize)> = Vec::new();
289
290        for n in &self.nodes {
291            let (entity, action) = (n.step.entity(), n.step.action());
292
293            match out
294                .iter_mut()
295                .find(|(e, a, _)| *e == entity && *a == action)
296            {
297                Some((_, _, count)) => *count += 1,
298                None => out.push((entity, action, 1)),
299            }
300        }
301
302        out
303    }
304
305    /// Executes up to `max_steps` steps (leaves first) in the transaction and returns the
306    /// number of steps processed. Commit the transaction between calls to release the lock.
307    ///
308    /// Fails without changing anything when the plan is restricted. If the transaction
309    /// fails to commit, build a new plan: the steps already done are no-ops.
310    pub async fn execute(
311        &mut self,
312        trx: &mut CtxTransaction<'_>,
313        max_steps: usize,
314    ) -> Result<usize> {
315        if let Some(r) = self.restrictions.first() {
316            return Err(r.to_error());
317        }
318
319        let mut linked = FxHashMap::default();
320        let mut done = self.done;
321        let end = done.saturating_add(max_steps).min(self.nodes.len());
322
323        while done < end {
324            let i = self.nodes.len() - 1 - done;
325
326            if self.is_linked(trx, i, &mut linked).await? {
327                self.nodes[i].step.apply(trx).await?;
328            }
329
330            done += 1;
331        }
332
333        let count = done - self.done;
334        self.done = done;
335        Ok(count)
336    }
337
338    /// Checks the foreign key chain from the node up to the root.
339    async fn is_linked(
340        &self,
341        trx: &CtxTransaction<'_>,
342        mut i: usize,
343        memo: &mut FxHashMap<usize, bool>,
344    ) -> Result<bool> {
345        let mut path = Vec::new();
346
347        let mut linked = loop {
348            if let Some(&v) = memo.get(&i) {
349                break v;
350            }
351
352            path.push(i);
353
354            match self.nodes[i].parent {
355                Some(p) => i = p,
356                None => break true,
357            }
358        };
359
360        for &n in path.iter().rev() {
361            linked = linked && self.nodes[n].step.is_linked(trx).await?;
362            memo.insert(n, linked);
363        }
364
365        Ok(linked)
366    }
367}

369/// Collects the children of the node being planned. Used by the planning handlers.
370pub struct DeletePlanner {
371    node: usize,
372    pending: Vec<Box<dyn PlanStep>>,
373    restrictions: Vec<DeleteRestriction>,
374    visited: FxHashMap<TypeId, Box<dyn Any + Send + Sync>>,
375}
376
377impl DeletePlanner {
378    /// Returns `false` if the entity is already planned for removal.
379    fn visit<E: Entity>(&mut self, key: &E::Key) -> bool {
380        self.visited
381            .entry(TypeId::of::<E>())
382            .or_insert_with(|| Box::new(FxHashSet::<E::Key>::default()))
383            .downcast_mut::<FxHashSet<E::Key>>()
384            .is_some_and(|set| set.insert(key.clone()))
385    }
386}

388trait PlanStep: Send + Sync {
389    fn action(&self) -> DeleteAction;
390    fn entity(&self) -> &'static str;
391    fn entity_type(&self) -> TypeId;
392    fn field(&self) -> Option<&'static str>;
393    fn key_any(&self) -> &(dyn Any + Send + Sync);
394    fn key_debug(&self) -> &dyn Debug;
395
396    /// Indicates if the entity still references its parent in the plan.
397    fn is_linked<'a>(&'a self, trx: &'a CtxTransaction<'_>) -> BoxFuture<'a, Result<bool>>;
398
399    fn apply<'a>(&'a self, trx: &'a mut CtxTransaction<'_>) -> BoxFuture<'a, Result<bool>>;
400}

443struct ChildStep<F: ForeignKey> {

449impl<F: ForeignKey> ChildStep<F>

455        trx: &'a CtxTransaction<'_>,

457        Ok(entity_of::<F::Child>(trx, &self.key)

459            .filter(|c| F::foreign_key(&self.key, c) == Some(&self.parent)))
460    }

514        DeleteAction::SetNull
515    }

517    fn entity(&self) -> &'static str {
518        F::Child::NAME
519    }
520
521    fn entity_type(&self) -> TypeId {
522        TypeId::of::<F::Child>()
523    }
524
525    fn field(&self) -> Option<&'static str> {
526        Some(F::FIELD)
527    }
528
529    fn key_any(&self) -> &(dyn Any + Send + Sync) {
530        &self.0.key
531    }
532
533    fn key_debug(&self) -> &dyn Debug {
534        &self.0.key
535    }
536
537    fn is_linked<'a>(&'a self, trx: &'a CtxTransaction<'_>) -> BoxFuture<'a, Result<bool>> {
538        Box::pin(async move { Ok(self.0.linked_child(trx).await?.is_some()) })
539    }
540
541    fn apply<'a>(&'a self, trx: &'a mut CtxTransaction<'_>) -> BoxFuture<'a, Result<bool>> {
542        Box::pin(async move {
543            let Some(child) = self.0.linked_child(trx).await? else {
544                return Ok(false);
545            };
546
547            let mut child = child.clone();
548            F::set_null(&mut child);
549            F::save(trx, self.0.key.clone(), child).await?;
550            Ok(true)
551        })
552    }

555/// Planning handler of the parent for [OnDelete::Cascade](crate::OnDelete::Cascade).
556pub fn plan_cascade<'a, F>(
557    trx: &'a mut CtxTransaction<'_>,
558    key: &'a (dyn Any + Send + Sync),
559    planner: &'a mut DeletePlanner,
560) -> BoxFuture<'a, Result<()>>
561where
562    F: ForeignKey,
563    F::Child: EntityRemove,
564    Ctx: AsRefAsync<<F::Child as EntityAccessor>::Tbl>,
565    ProviderContainer: LoadAll<F::Child, (), <F::Child as EntityAccessor>::Tbl>,
566    for<'b> TransactionProvider<'b>: Delete<F::Child>,
567{
568    Box::pin(async move {
569        let Some(parent) = key.downcast_ref::<ParentKey<F>>() else {
570            return Ok(());
571        };
572
573        for key in children_of::<F>(trx, parent).await? {
574            if planner.visit::<F::Child>(&key) {
575                planner.pending.push(Box::new(Cascade::<F>(ChildStep {
576                    parent: parent.clone(),
577                    key,
578                    _f: PhantomData,
579                })));
580            }
581        }
582
583        Ok(())
584    })
585}

587/// Planning handler of the parent for [OnDelete::Restrict](crate::OnDelete::Restrict).
588pub fn plan_restrict<'a, F>(
589    trx: &'a mut CtxTransaction<'_>,
590    key: &'a (dyn Any + Send + Sync),
591    planner: &'a mut DeletePlanner,
592) -> BoxFuture<'a, Result<()>>
593where
594    F: ForeignKey,
595    Ctx: AsRefAsync<<F::Child as EntityAccessor>::Tbl>,
596    ProviderContainer: LoadAll<F::Child, (), <F::Child as EntityAccessor>::Tbl>,
597{
598    Box::pin(async move {
599        let Some(parent) = key.downcast_ref::<ParentKey<F>>() else {
600            return Ok(());
601        };
602
603        let count = count_children::<F>(trx, parent).await?;
604
605        if count > 0 {
606            planner.restrictions.push(DeleteRestriction::ForeignKey {
607                node: planner.node,
608                entity: F::Child::NAME,
609                field: F::FIELD,
610                parent: F::Parent::NAME,
611                key: debug_string(parent),
612                count,
613            });
614        }
615
616        Ok(())
617    })
618}
