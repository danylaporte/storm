1//! Business rules restricting the removal of entities.
2//!
3//! A rule returns `Ok(Some(reason))` to block the removal. Rules are checked by
4//! [EntityRemove::remove](crate::EntityRemove::remove) (including cascaded removals) before
5//! anything is modified, and reported by [DeletePlan](crate::DeletePlan) as restrictions.
6//!
7//! ```ignore
8//! #[storm::register]
9//! fn rules() {
10//!     storm::on_remove_rule::<Invoice>(posted_invoice);
11//! }
12//!
13//! fn posted_invoice<'a>(
14//!     trx: &'a mut CtxTransaction<'_>,
15//!     key: &'a InvoiceId,
16//! ) -> BoxFuture<'a, Result<Option<String>>> {
17//!     Box::pin(async move {
18//!         let posted = trx.get_entity::<Invoice>(key).await?.is_some_and(|i| i.posted);
19//!         Ok(posted.then(|| "a posted invoice cannot be removed".into()))
20//!     })
21//! }
22//! ```
23
24use crate::{
25    BoxFuture, CtxTransaction, CtxTypeInfo, Entity, Error, Result, error::debug_string,
26    registry::InitCell,
27};
28use rustc_hash::{FxBuildHasher, FxHashMap};
29use std::{
30    any::{Any, TypeId},
31    fmt::Debug,
32};
33
34pub type RemoveRuleFn<E> = for<'a> fn(
35    trx: &'a mut CtxTransaction<'_>,
36    key: &'a <E as Entity>::Key,
37) -> BoxFuture<'a, Result<Option<String>>>;
38
39type ErasedFn = for<'a> fn(
40    rule: &'a (dyn Any + Send + Sync),
41    trx: &'a mut CtxTransaction<'_>,
42    key: &'a (dyn Any + Send + Sync),
43) -> BoxFuture<'a, Result<Option<String>>>;
44
45pub(crate) struct RemoveRule {
46    pub(crate) entity: &'static str,
47    rule: Box<dyn Any + Send + Sync>,
48    call: ErasedFn,
49}
50
51impl RemoveRule {
52    /// Returns the reason when the removal of `key` is blocked.
53    pub(crate) fn check<'a>(
54        &'static self,
55        trx: &'a mut CtxTransaction<'_>,
56        key: &'a (dyn Any + Send + Sync),
57    ) -> BoxFuture<'a, Result<Option<String>>> {
58        (self.call)(&*self.rule, trx, key)
59    }
60}
61
62/// Rules by entity type, in registration order.
63static RULES: InitCell<FxHashMap<TypeId, Vec<RemoveRule>>> =
64    InitCell::new(FxHashMap::with_hasher(FxBuildHasher));
65
66/// Registers a business rule restricting the removal of `E`. Call in a `#[storm::register]` fn.
67pub fn on_remove_rule<E: Entity + CtxTypeInfo>(f: RemoveRuleFn<E>) {
68    let rules = RULES.get_mut().entry(TypeId::of::<E>()).or_default();
69
70    rules.push(RemoveRule {
71        entity: E::NAME,
72        rule: Box::new(f),
73        call: call_erased::<E>,
74    });
75}
76
77fn call_erased<'a, E: Entity>(
78    rule: &'a (dyn Any + Send + Sync),
79    trx: &'a mut CtxTransaction<'_>,
80    key: &'a (dyn Any + Send + Sync),
81) -> BoxFuture<'a, Result<Option<String>>> {
82    match (rule.downcast_ref::<RemoveRuleFn<E>>(), key.downcast_ref()) {
83        (Some(f), Some(key)) => f(trx, key),
84        _ => Box::pin(async { Ok(None) }),
85    }
86}
87
88#[inline]
89pub(crate) fn rules_of(ty: TypeId) -> &'static [RemoveRule] {
90    RULES.get().get(&ty).map_or(&[], Vec::as_slice)
91}
92
93/// Fails with [Error::RemoveRestricted] on the first rule blocking the removal of `key`.
94///
95/// Not generic so its state machine is compiled once instead of once per entity.
96pub(crate) async fn check(
97    trx: &mut CtxTransaction<'_>,
98    ty: TypeId,
99    key: &(dyn Any + Send + Sync),
100    key_dbg: &(dyn Debug + Sync),
101) -> Result<()> {
102    for rule in rules_of(ty) {
103        if let Some(reason) = rule.check(trx, key).await? {
104            return Err(Error::RemoveRestricted {
105                entity: rule.entity,
106                key: debug_string(key_dbg),
107                reason,
108            });
109        }
110    }
111
112    Ok(())
113}
