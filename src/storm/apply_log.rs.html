1use crate::{Ctx, CtxExt, Logs, registry::InitCell};
2use extobj::VarId;
3use rustc_hash::{FxBuildHasher, FxHashMap};
4use std::cmp::Ordering;
5
6pub trait ApplyLog<L> {
7    fn apply_log(&mut self, log: L) -> bool;
8}
9
10type ApplierFn = fn(ctx: &mut Ctx, logs: &mut Logs) -> bool;
11
12struct Applier {
13    f: ApplierFn,
14    order: ApplyOrder,
15    triggers: Box<[VarId<CtxExt>]>,
16}
17
18struct Appliers {
19    /// Sorted by `ApplyOrder`.
20    list: Vec<Applier>,
21
22    /// For each log var, the ascending positions in `list` of the appliers it triggers.
23    by_var: FxHashMap<VarId<CtxExt>, Box<[u32]>>,
24}
25
26static APPLIERS: InitCell<Appliers> = InitCell::new(Appliers {
27    list: Vec::new(),
28    by_var: FxHashMap::with_hasher(FxBuildHasher),
29});
30
31/// Private: use by a macro.
32///
33/// `triggers` are the log vars consumed by the applier; it is only invoked when at
34/// least one of them is present in the logs being applied.
35#[doc(hidden)]
36pub fn __register_apply(f: ApplierFn, order: ApplyOrder, triggers: &[VarId<CtxExt>]) {
37    let list = &mut APPLIERS.get_mut().list;
38
39    let (Ok(index) | Err(index)) = list.binary_search_by_key(&order, |a| a.order);
40
41    list.insert(
42        index,
43        Applier {
44            f,
45            order,
46            triggers: triggers.into(),
47        },
48    );
49}
50
51/// Builds the var → appliers lookup. Must be called at the end of the registration phase.
52pub(crate) fn finalize_appliers() {
53    let appliers = APPLIERS.get_mut();
54    let mut by_var = FxHashMap::<VarId<CtxExt>, Vec<u32>>::default();
55
56    for (index, applier) in appliers.list.iter().enumerate() {
57        let index = u32::try_from(index).expect("too many appliers");
58
59        for var in &applier.triggers {
60            let positions = by_var.entry(*var).or_default();
61
62            if positions.last() != Some(&index) {
63                positions.push(index);
64            }
65        }
66    }
67
68    appliers.by_var = by_var.into_iter().map(|(k, v)| (k, v.into())).collect();
69}
70
71pub(crate) fn perform_apply_log(ctx: &mut Ctx, mut logs: Logs) -> bool {
72    let appliers = APPLIERS.get();
73    let triggered = |var| appliers.by_var.get(&var).map_or(&[][..], |p| &p[..]);
74    let mut changed = false;
75
76    match logs.len() {
77        0 => {}
78        1 => {
79            let var = logs.var_ids().next().expect("one log");
80
81            for &index in triggered(var) {
82                changed |= (appliers.list[index as usize].f)(ctx, &mut logs);
83            }
84        }
85        _ => {
86            // Merge the triggered appliers of every log, keeping the global apply order.
87            let mut positions = logs
88                .var_ids()
89                .flat_map(triggered)
90                .copied()
91                .collect::<Vec<_>>();
92
93            positions.sort_unstable();
94            positions.dedup();
95
96            for index in positions {
97                changed |= (appliers.list[index as usize].f)(ctx, &mut logs);
98            }
99        }
100    }
101
102    changed
103}

105#[derive(Clone, Copy, Debug)]
106#[repr(u8)]
107pub enum ApplyOrder {
108    FlatSet = 5,
109    NodeSet = 10,
110    Tree = 15,
111    Table = 20,
112}
113
114impl Eq for ApplyOrder {}
115
116impl Ord for ApplyOrder {
117    #[inline]
118    fn cmp(&self, other: &Self) -> Ordering {
119        (*self as u8).cmp(&(*other as u8))
120    }
121}
122
123impl PartialOrd for ApplyOrder {
124    #[inline]
125    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
126        Some(self.cmp(other))
127    }
128}
129
130impl PartialEq for ApplyOrder {
131    #[inline]
132    fn eq(&self, other: &Self) -> bool {
