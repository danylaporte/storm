18/// Private: used by macros.
19///
20/// Resolves the field of an entity by its PascalCase name, or `None` when the entity has no
21/// fields enum (autoref specialization over [`__FieldOfEntity`] and [`__FieldOfNone`]).
22#[doc(hidden)]
23pub struct __FieldOf<E>(PhantomData<E>);
24
25impl<E> __FieldOf<E> {
26    #[allow(clippy::new_without_default)]
27    #[inline]
28    pub const fn new() -> Self {
29        Self(PhantomData)
30    }
31}

33#[doc(hidden)]
34pub trait __FieldOfEntity {
35    fn __field(&self, name: &str) -> Option<Box<dyn Fields>>;
36}
37
38impl<E> __FieldOfEntity for __FieldOf<E>
39where
40    E: EntityFields,
41    E::Fields: Fields,
42{
43    #[inline]
44    fn __field(&self, name: &str) -> Option<Box<dyn Fields>> {
45        E::field_by_name(name).map(|f| Box::new(f) as Box<dyn Fields>)
46    }
47}

50pub trait __FieldOfNone {

53        None
54    }

66/// Deserializes the fields enum generated by `MssqlSave` and returns the index of the variant.
67///
68/// Accepts the same input as `#[derive(Deserialize)]` on a unit-only enum, but is compiled once
69/// instead of once per entity.
70#[doc(hidden)]
71pub fn _deserialize_fields<'de, D: Deserializer<'de>>(
72    deserializer: D,
73    name: &'static str,
74    variants: &'static [&'static str],
75) -> Result<usize, D::Error> {
76    deserializer.deserialize_enum(name, variants, FieldsVisitor { name, variants })
77}
78
79struct FieldsVisitor {
80    name: &'static str,
81    variants: &'static [&'static str],
82}
83
84impl<'de> Visitor<'de> for FieldsVisitor {
85    type Value = usize;
86
87    fn expecting(&self, f: &mut fmt::Formatter) -> fmt::Result {
88        write!(f, "enum {}", self.name)
89    }
90
91    fn visit_enum<A: EnumAccess<'de>>(self, data: A) -> Result<usize, A::Error> {
92        let (index, variant) = data.variant_seed(VariantIdentifier(self.variants))?;
93        variant.unit_variant()?;
94        Ok(index)
95    }
96}

98struct VariantIdentifier(&'static [&'static str]);
99
100impl<'de> DeserializeSeed<'de> for VariantIdentifier {
101    type Value = usize;
102
103    fn deserialize<D: Deserializer<'de>>(self, deserializer: D) -> Result<usize, D::Error> {
104        deserializer.deserialize_identifier(self)
105    }
106}
107
108impl Visitor<'_> for VariantIdentifier {
109    type Value = usize;
110
111    fn expecting(&self, f: &mut fmt::Formatter) -> fmt::Result {
112        f.write_str("variant identifier")
113    }
114
115    fn visit_u64<E: de::Error>(self, v: u64) -> Result<usize, E> {
116        usize::try_from(v)
117            .ok()
118            .filter(|&i| i < self.0.len())
119            .ok_or_else(|| {
120                let msg = format!("variant index 0 <= i < {}", self.0.len());
121                E::invalid_value(Unexpected::Unsigned(v), &msg.as_str())
122            })
123    }
124
125    fn visit_str<E: de::Error>(self, v: &str) -> Result<usize, E> {
126        self.0
127            .iter()
128            .position(|n| *n == v)
129            .ok_or_else(|| E::unknown_variant(v, self.0))
130    }
131
132    fn visit_bytes<E: de::Error>(self, v: &[u8]) -> Result<usize, E> {
133        match std::str::from_utf8(v) {
134            Ok(s) => self.visit_str(s),
135            Err(_) => Err(E::unknown_variant(&String::from_utf8_lossy(v), self.0)),
136        }
137    }
138}

142mod tests {
143    use super::{_deserialize_fields, FieldsOrStr};

146    #[derive(Clone, Copy, Debug, Deserialize, PartialEq)]
147    #[serde(rename_all = "camelCase")]
148    enum Derived {
149        FirstName,
150        Age,
151    }
152
153    #[derive(Clone, Copy, Debug, PartialEq)]
154    enum Manual {
155        FirstName,
156        Age,
157    }
158
159    impl<'de> Deserialize<'de> for Manual {
160        fn deserialize<D: Deserializer<'de>>(d: D) -> Result<Self, D::Error> {
161            // Same name as the derived enum so the error messages are comparable.
162            let i = _deserialize_fields(d, "Derived", &["firstName", "age"])?;
163            Ok([Self::FirstName, Self::Age][i])
164        }
165    }

167    fn same(json: &str) {
168        let as_u8 = |r: Result<u8, String>| r;
169
170        // Directly from the text.
171        let d = serde_json::from_str::<Derived>(json)
172            .map(|v| v as u8)
173            .map_err(|e| e.to_string());
174        let m = serde_json::from_str::<Manual>(json)
175            .map(|v| v as u8)
176            .map_err(|e| e.to_string());
177        assert_eq!(as_u8(d), as_u8(m), "text {json}");
178
179        // From a `Value`, another `Deserializer`.
180        if let Ok(value) = serde_json::from_str::<serde_json::Value>(json) {
181            let d = Derived::deserialize(&value)
182                .map(|v| v as u8)
183                .map_err(|e| e.to_string());
184            let m = Manual::deserialize(&value)
185                .map(|v| v as u8)
186                .map_err(|e| e.to_string());
187            assert_eq!(d, m, "value {json}");
188        }
189
190        // Buffered by `#[serde(untagged)]`.
191        let d = serde_json::from_str::<FieldsOrStr<Derived>>(json)
192            .map(|v| match v {
193                FieldsOrStr::Fields(f) => Ok(f as u8),
194                FieldsOrStr::String(s) => Err(s),
195            })
196            .map_err(|e| e.to_string());
197        let m = serde_json::from_str::<FieldsOrStr<Manual>>(json)
198            .map(|v| match v {
199                FieldsOrStr::Fields(f) => Ok(f as u8),
200                FieldsOrStr::String(s) => Err(s),
201            })
202            .map_err(|e| e.to_string());
203        assert_eq!(d, m, "untagged {json}");
204    }
